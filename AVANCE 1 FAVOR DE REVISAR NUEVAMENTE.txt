    Definir proyectos_registrados Como cadena

    SECCIÓN 1: FUNCIONES AUXILIARES Y MATEMÁTICAS
    1
    1.1 SubProceso calcular_restante (inicial, gastado)
    1.2     Retornar inicial - gastado
    1.3 FinSubProceso

    1.4 SubProceso calcular_porcentaje (cumplidas, total)
    1.5     Retornar (cumplidas / total) * 100
    1.6 FinSubProceso

    1.7 SubProceso calcular_semanas (dias)
    1.8     Retornar TRUNC(dias / 7)
    1.9 FinSubProceso

    1.10 SubProceso calcular_dias_sobrantes (dias, semanas)
    1.11    Retornar dias - (semanas * 7)
    1.12 FinSubProceso

    1.13 SubProceso generar_barra_progreso (porcentaje)
    1.14    Definir bloques Como Entero
    1.15    Si porcentaje < 0 Entonces
    1.16        porcentaje <- 0
    1.17    FinSi
    1.18    Si porcentaje > 100 Entonces
    1.19        porcentaje <- 100
    1.20    FinSi
    1.21    bloques <- TRUNC(porcentaje / 10)
    1.22    Escribir "Progreso: [", bloques, " de 10 bloques completados]"
    1.23 FinSubProceso

    1.24 SubProceso evaluar_salud_presupuesto (inicial, gastado)
    1.25    Definir porcentaje_gastado como real
    1.26    Si inicial > 0 Entonces
    1.27        porcentaje_gastado <- (gastado / inicial) * 100
    1.28    Sino
    1.29        porcentaje_gastado <- 0
    1.30    FinSi
    1.31    Si porcentaje_gastado > 90 Entonces
    1.32        Escribir "Alerta: Presupuesto casi agotado (Mayor al 90%)"
    1.33    Sino
    1.34        Si porcentaje_gastado >= 70 Entonces
    1.35            Escribir "Precaución: Presupuesto en riesgo moderado (Entre 70% y 90%)"
    1.36        Sino
    1.37            Escribir "Saludable: Presupuesto bajo control (Menor al 70%)"
    1.38        FinSi
    1.39    FinSi
    1.40 FinSubProceso

    SECCIÓN 3: REGISTRO Y MÉTRICAS DE PROYECTOS
    3
    3.1 SubProceso registrar_proyecto()
    3.2     Definir nombre como cadena
    3.3     Definir avance como real
    3.4     Definir continuar como cadena
    3.5     continuar <- "s"
    3.6     Mientras continuar = "s" O continuar = "S" Hacer
    3.7         Escribir "REGISTRO DE PROYECTO"
    3.8         Escribir "Nombre del proyecto:"
    3.9         Leer nombre
    3.10        Escribir "Porcentaje de avance (0 a 100):"
    3.11        Leer avance
    3.12        Si avance >= 0 Y avance <= 100 Entonces
    3.13            Escribir "Proyecto ", nombre, " registrado con ", avance, "% de avance"
    3.14            generar_barra_progreso(avance)
    3.15            AGREGAR_A_LISTA(proyectos_registrados, nombre + " - " + CONVERTIR_A_TEXTO(avance) + "%")
    3.16            AGREGAR_A_MATRIZ(matriz_proyectos, nombre, CONVERTIR_A_TEXTO(avance) + "%", "Activo")
    3.17        Sino
    3.18            Escribir "El porcentaje debe estar entre 0 y 100"
    3.19        FinSi
    3.20        Escribir "¿Desea registrar otro proyecto? (s/n):"
    3.21        Leer continuar
    3.22    FinMientras
    3.23 FinSubProceso

    3.24 SubProceso calcular_metricas()
    3.25    Definir nombre como cadena
    3.26    Definir metas_total, metas_cumplidas Como Entero
    3.27    Definir presupuesto_inicial, presupuesto_gastado Como Real
    3.28    Definir dias, semanas, dias_s Como Entero
    3.29    Definir presupuesto_restante, porcentaje_avance Como Real
    3.30    Escribir "CALCULO DE METRICAS DE FINANZAS Y TIEMPO"
    3.31    Escribir "Nombre del proyecto:"
    3.32    Leer nombre
    3.33    Escribir "Total de metas:"
    3.34    Leer metas_total
    3.35    Escribir "Metas alcanzadas:"
    3.36    Leer metas_cumplidas
    3.37    Escribir "Presupuesto inicial:"
    3.38    Leer presupuesto_inicial
    3.39    Escribir "Presupuesto gastado:"
    3.40    Leer presupuesto_gastado
    3.41    Escribir "Duracion estimada en dias del proyecto:"
    3.42    Leer dias
    3.43    Si metas_total <= 0 Entonces
    3.44        Escribir "Error: Sus metas deben ser al menos 1"
    3.45    Sino
    3.46        presupuesto_restante <- calcular_restante(presupuesto_inicial, presupuesto_gastado)
    3.47        porcentaje_avance <- calcular_porcentaje(metas_cumplidas, metas_total)
    3.48        semanas <- calcular_semanas(dias)
    3.49        dias_s <- calcular_dias_sobrantes(dias, semanas)
    3.50        Escribir "RESUMEN DE RESULTADOS"
    3.51        Escribir "Resumen de metricas: ", nombre
    3.52        Escribir "Avance real de metas: ", porcentaje_avance, "%"
    3.53        generar_barra_progreso(porcentaje_avance)
    3.54        evaluar_salud_presupuesto(presupuesto_inicial, presupuesto_gastado)
    3.55        Escribir "Presupuesto disponible: $", presupuesto_restante
    3.56        Escribir "Tiempo estimado: ", semanas, " semanas y ", dias_s, " dias"
    3.57    FinSi
    3.58 FinSubProceso
