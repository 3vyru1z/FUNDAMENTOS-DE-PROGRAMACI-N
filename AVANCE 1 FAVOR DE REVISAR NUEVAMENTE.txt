    SECCIÓN 1: FUNCIONES AUXILIARES Y MATEMÁTICAS
    1
    1.1 SubProceso calcular_restante (inicial, gastado)
    1.2     Retornar inicial - gastado
    1.3 FinSubProceso

    1.4 SubProceso calcular_porcentaje (cumplidas, total)
    1.5     Retornar (cumplidas / total) * 100
    1.6 FinSubProceso

    1.7 SubProceso calcular_semanas (dias)
    1.8     Retornar TRUNC(dias / 7)
    1.9 FinSubProceso

    1.10 SubProceso calcular_dias_sobrantes (dias, semanas)
    1.11    Retornar dias - (semanas * 7)
    1.12 FinSubProceso

    1.13 SubProceso generar_barra_progreso (porcentaje)
    1.14    Definir bloques como entero
    1.15    Si porcentaje < 0 Entonces
    1.16        porcentaje <-- 0
    1.17    FinSi
    1.18    Si porcentaje > 100 Entonces
    1.19        porcentaje <-- 100
    1.20    FinSi
    1.21    bloques <-- TRUNC(porcentaje / 10)
    1.22    Escribir "Progreso: [", bloques, " de 10 bloques completados]"
    1.23 FinSubProceso

    1.24 SubProceso evaluar_salud_presupuesto (inicial, gastado)
    1.25    Definir porcentaje_gastado como real
    1.26    Si inicial > 0 Entonces
    1.27        porcentaje_gastado <-- (gastado / inicial) * 100
    1.28    Sino
    1.29        porcentaje_gastado <-- 0
    1.30    FinSi
    1.31    Si porcentaje_gastado > 90 Entonces
    1.32        Escribir "Alerta: Presupuesto casi agotado (Mayor al 90%)"
    1.33    Sino
    1.34        Si porcentaje_gastado >= 70 Entonces
    1.35            Escribir "Precaución: Presupuesto en riesgo moderado (Entre 70% y 90%)"
    1.36        Sino
    1.37            Escribir "Saludable: Presupuesto bajo control (Menor al 70%)"
    1.38        FinSi
    1.39    FinSi
    1.40 FinSubProceso

    SECCIÓN 2: MÓDULO DE AUTENTICACIÓN
    2
    2.1 SubProceso crear_cuenta()
    2.2     Definir nuevo_usuario, nueva_contrasena como una cadena
    2.3     Escribir "REGISTRO NUEVA CUENTA"
    2.4     Escribir "Ingrese un nombre de usuario:"
    2.5     Leer nuevo_usuario
    2.6     Escribir "Ingrese una contrasena:"
    2.7     Leer nueva_contrasena
    2.8     Si nuevo_usuario <> "" Y nueva_contrasena <> "" Entonces
    2.9         Escribir "Cuenta registrada con exito para: ", nuevo_usuario
    2.10    Sino
    2.11        Escribir "Error: Debe llenar todos los campos"
    2.12    FinSi
    2.13 FinSubProceso

    2.14 SubProceso iniciar_sesion()
    2.15    Definir usuario, contrasena como una cadena
    2.16    Escribir "INICIO DE SESION"
    2.17    Escribir "Ingrese su usuario:"
    2.18    Leer usuario
    2.19    Escribir "Ingrese contrasena:"
    2.20    Leer contrasena
    2.21    Si usuario <> "" Y contrasena <> "" Entonces
    2.22        Escribir "Bienvenido/a, ", usuario
    2.23    Sino
    2.24        Escribir "Datos incorrectos"
    2.25    FinSi
    2.26 FinSubProceso

    SECCIÓN 3: REGISTRO Y MÉTRICAS DE PROYECTOS
    3
    3.1 SubProceso registrar_proyecto()
    3.2     Definir nombre como una cadena
    3.3     Definir avance como real
    3.4     Definir continuar como una cadena
    3.5     Definir estado_inicial como una cadena
    3.6     continuar <-- "s"
    3.7     Mientras continuar = "s" O continuar = "S" Hacer
    3.8         Escribir "REGISTRO DE PROYECTO"
    3.9         Escribir "Nombre del proyecto:"
    3.10        Leer nombre
    3.11        Escribir "Porcentaje de avance (0 a 100):"
    3.12        Leer avance
    3.13        Si avance >= 0 Y avance <-- 100 Entonces
    3.14            Escribir "Proyecto ", nombre, " registrado con ", avance, "% de avance"
    3.15            generar_barra_progreso(avance)
    3.16            Si avance = 100 Entonces
    3.17                estado_inicial <-- "Completado"
    3.18            Sino
    3.19                Si avance > 0 Entonces
    3.20                    estado_inicial <-- "En Proceso"
    3.21                Sino
    3.22                    estado_inicial <-- "Iniciado"
    3.23                FinSi
    3.24            FinSi
    3.25            AGREGAR_A_LISTA(proyectos_registrados, nombre + " - " + CONVERTIR_A_TEXTO(avance) + "%")
    3.26            AGREGAR_A_MATRIZ(matriz_proyectos, nombre, CONVERTIR_A_TEXTO(avance) + "%", estado_inicial)
    3.27        Sino
    3.28            Escribir "El porcentaje debe estar entre 0 y 100"
    3.29        FinSi
    3.30        Escribir "¿Desea registrar otro proyecto? (s/n):"
    3.31        Leer continuar
    3.32    FinMientras
    3.33 FinSubProceso

    3.34 SubProceso calcular_metricas()
    3.35    Definir nombre como una cadena
    3.36    Definir metas_total, metas_cumplidas como entero
    3.37    Definir presupuesto_inicial, presupuesto_gastado Como Real
    3.38    Definir dias, semanas, dias_s como entero
    3.39    Definir presupuesto_restante, porcentaje_avance Como Real
    3.40    Escribir "CALCULO DE METRICAS DE FINANZAS Y TIEMPO"
    3.41    Escribir "Nombre del proyecto:"
    3.42    Leer nombre
    3.43    Escribir "Total de metas:"
    3.44    Leer metas_total
    3.45    Escribir "Metas alcanzadas:"
    3.46    Leer metas_cumplidas
    3.47    Escribir "Presupuesto inicial:"
    3.48    Leer presupuesto_inicial
    3.49    Escribir "Presupuesto gastado:"
    3.50    Leer presupuesto_gastado
    3.51    Escribir "Duracion estimada en dias del proyecto:"
    3.52    Leer dias
    3.53    Si metas_total <-- 0 Entonces
    3.54        Escribir "Error: Sus metas deben ser al menos 1"
    3.55    Sino
    3.56        presupuesto_restante <-- calcular_restante(presupuesto_inicial, presupuesto_gastado)
    3.57        porcentaje_avance <-- calcular_porcentaje(metas_cumplidas, metas_total)
    3.58        semanas <-- calcular_semanas(dias)
    3.59        dias_s <-- calcular_dias_sobrantes(dias, semanas)
    3.60        Escribir "RESUMEN DE RESULTADOS"
    3.61        Escribir "Resumen de metricas: ", nombre
    3.62        Escribir "Avance real de metas: ", porcentaje_avance, "%"
    3.63        generar_barra_progreso(porcentaje_avance)
    3.64        evaluar_salud_presupuesto(presupuesto_inicial, presupuesto_gastado)
    3.65        Escribir "Presupuesto disponible: $", presupuesto_restante
    3.66        Escribir "Tiempo estimado: ", semanas, " semanas y ", dias_s, " dias"
    3.67    FinSi
    3.68 FinSubProceso

    SECCIÓN 4: LISTAS, MATRICES Y ANÁLISIS DE CADENAS
    4
    4.1 SubProceso ver_historial_proyectos()
    4.2     Escribir "HISTORIAL DE PROYECTOS REGISTRADOS"
    4.3     Si LONGITUD_LISTA(proyectos_registrados) = 0 Entonces
    4.4         Escribir "No hay proyectos registrados aun."
    4.5     Sino
    4.6         Para cada elemento en proyectos_registrados Hacer
    4.7             Escribir "- ", elemento
    4.8         FinPara
    4.9     FinSi
    4.10 FinSubProceso

    4.11 SubProceso calcular_estadisticas_lista()
    4.12    Definir total_elementos como entero
    4.13    Definir suma_avances, promedio_avance, mayor_avance, menor_avance, valor_actual Como Real
    4.14    Escribir "ESTADISTICAS GENERALES DE LA LISTA DE PROYECTOS"
    4.15    total_elementos <-- LONGITUD_LISTA(proyectos_registrados)
    4.16    Si total_elementos = 0 Entonces
    4.17        Escribir "La lista esta vacia."
    4.18    Sino
    4.19        suma_avances <-- 0
    4.20        mayor_avance <-- 0
    4.21        menor_avance <-- 100
    4.22        Para cada fila en matriz_proyectos hacer
    4.23            valor_actual <-- CONVERTIR_A_NUMERO(QUITAR_CARACTER(fila[1], "%"))
    4.24            suma_avances <-- suma_avances + valor_actual
    4.25            Si valor_actual > mayor_avance Entonces
    4.26                mayor_avance <-- valor_actual
    4.27            FinSi
    4.28            Si valor_actual < menor_avance Entonces
    4.29                menor_avance <-- valor_actual
    4.30            FinSi
    4.31        FinPara
    4.32        promedio_avance <-- suma_avances / total_elementos
    4.33        Escribir "Cantidad total de proyectos en el arreglo: ", total_elementos
    4.34        Escribir "Promedio general de avance: ", promedio_avance, "%"
    4.35        Escribir "Mayor avance registrado: ", mayor_avance, "%"
    4.36        Escribir "Menor avance registrado: ", menor_avance, "%"
    4.37    FinSi
    4.38 FinSubProceso

    4.39 SubProceso ver_matriz_proyectos()
    4.40    Escribir "TABLA DE DATOS - MATRIZ DE PROYECTOS"
    4.41    Si LONGITUD_LISTA(matriz_proyectos) = 0 Entonces
    4.42        Escribir "No hay datos en la matriz de proyectos."
    4.43    Sino
    4.44        Para cada fila en matriz_proyectos Hacer
    4.45            Escribir "Proyecto: ", fila[0], " - Avance: ", fila[1], " - Estado: ", fila[2]
    4.46        FinPara
    4.47    FinSi
    4.48 FinSubProceso

    4.49 SubProceso actualizar_estados_matriz()
    4.50    Definir avance_num como real
    4.51    Escribir "ACTUALIZACION DE ESTADOS EN LOTE"
    4.52    Para cada fila en matriz_proyectos Hacer
    4.53        avance_num <-- CONVERTIR_A_NUMERO(QUITAR_CARACTER(fila[1], "%"))
    4.54        Si avance_num = 100 Entonces
    4.55            fila[2] <-- "Completado"
    4.56        Sino
    4.57            Si avance_num > 0 Entonces
    4.58                fila[2] <-- "En Proceso"
    4.59            Sino
    4.60                fila[2] <-- "Iniciado"
    4.61            FinSi
    4.62        FinSi
    4.63        Escribir "Proyecto: ", fila[0], " -> Estado actualizado: ", fila[2]
    4.64    FinPara
    4.65 FinSubProceso

    4.66 SubProceso resumen_conteo_matriz()
    4.67    Definir iniciados, en_proceso, completados Como Entero
    4.68    iniciados <-- 0
    4.69    en_proceso <-- 0
    4.70    completados <-- 0
    4.71    Escribir "RESUMEN Y CONTEO CRUZADO DE PROYECTOS"
    4.72    Para cada fila en matriz_proyectos Hacer
    4.73        Si fila[2] = "Iniciado" Entonces
    4.74            iniciados <-- iniciados + 1
    4.75        Sino
    4.76            Si fila[2] = "En Proceso" Entonces
    4.77                en_proceso <-- en_proceso + 1
    4.78            Sino
    4.79                Si fila[2] = "Completado" Entonces
    4.80                    completados <-- completados + 1
    4.81                FinSi
    4.82            FinSi
    4.83        FinSi
    4.84    FinPara
    4.85    Escribir "Total de registros: ", LONGITUD_LISTA(matriz_proyectos)
    4.86    Escribir "Iniciados: ", iniciados
    4.87    Escribir "En Proceso: ", en_proceso
    4.88    Escribir "Completados: ", completados
    4.89 FinSubProceso

    4.90 SubProceso buscar_analizar_cadenas()
    4.91    Definir palabra_clave como una cadena
    4.92    Definir encontrados como entero
    4.93    encontrados <-- 0
    4.94    Escribir "BUSQUEDA Y ANALISIS DE CADENAS EN PROYECTOS"
    4.95    Escribir "Ingrese una palabra clave para buscar:"
    4.96    Leer palabra_clave
    4.97    palabra_clave <-- MINUSCULAS(RECORTAR(palabra_clave))
    4.98    Para cada elemento en proyectos_registrados Hacer
    4.99        Si CONTIENE(MINUSCULAS(elemento), palabra_clave) Entonces
    5.00            encontrados <-- encontrados + 1
    5.01            Escribir "Coincidencia [", encontrados, "]: ", elemento
    5.02            Escribir "En mayusculas: ", MAYUSCULAS(elemento)
    5.03            Escribir "Total de caracteres: ", LONGITUD_CADENA(elemento)
    5.04        FinSi
    5.05    FinPara
    5.06    Si encontrados = 0 Entonces
    5.07        Escribir "No se encontraron proyectos con esa palabra."
    5.08    Sino
    5.09        Escribir "Total de coincidencias: ", encontrados
    5.10    FinSi
    5.11 FinSubProceso

    SECCIÓN 5: MANEJO DE ARCHIVOS
    5
    5.1 SubProceso guardar_datos_archivos()
    5.2     Escribir "GUARDAR PROYECTOS Y MATRIZ EN ARCHIVO"
    5.3     ABRIR_ARCHIVO_ESCRITURA(archivo_proyectos)
    5.4     Para cada elemento en proyectos_registrados Hacer
    5.5         ESCRIBIR_LINEA(archivo_proyectos, elemento)
    5.6     FinPara
    5.7     CERRAR_ARCHIVO(archivo_proyectos)
    5.8     ABRIR_ARCHIVO_ESCRITURA(archivo_matriz)
    5.9     Para cada fila en matriz_proyectos Hacer
    5.10        ESCRIBIR_LINEA(archivo_matriz, fila[0] + "-" + fila[1] + "-" + fila[2])
    5.11    FinPara
    5.12    CERRAR_ARCHIVO(archivo_matriz)
    5.13    Escribir "Datos guardados en ", archivo_proyectos, " y ", archivo_matriz
    5.14 FinSubProceso

    5.15 SubProceso cargar_datos_archivos()
    5.16    Definir linea, linea_m como una cadena
    5.17    Definir campos como una cadena
    5.18    Escribir "CARGAR PROYECTOS Y MATRIZ DESDE ARCHIVO"
    5.19    Si EXISTE_ARCHIVO(archivo_proyectos) Entonces
    5.20        ABRIR_ARCHIVO_LECTURA(archivo_proyectos)
    5.21        LIMPIAR_LISTA(proyectos_registrados)
    5.22        Mientras NO FIN_ARCHIVO(archivo_proyectos) Hacer
    5.23            linea <-- RECORTAR(LEER_LINEA(archivo_proyectos))
    5.24            Si linea <> "" Entonces
    5.25                AGREGAR_A_LISTA(proyectos_registrados, linea)
    5.26            FinSi
    5.27        FinMientras
    5.28        CERRAR_ARCHIVO(archivo_proyectos)
    5.29        Si EXISTE_ARCHIVO(archivo_matriz) Entonces
    5.30            ABRIR_ARCHIVO_LECTURA(archivo_matriz)
    5.31            LIMPIAR_LISTA(matriz_proyectos)
    5.32            Mientras NO FIN_ARCHIVO(archivo_matriz) Hacer
    5.33                linea_m <-- RECORTAR(LEER_LINEA(archivo_matriz))
    5.34                Si linea_m <> "" Entonces
    5.35                    campos <-- DIVIDIR_TEXTO(linea_m, "-")
    5.36                    AGREGAR_A_MATRIZ(matriz_proyectos, campos[0], campos[1], campos[2])
    5.37                FinSi
    5.38            FinMientras
    5.39            CERRAR_ARCHIVO(archivo_matriz)
    5.40        FinSi
    5.41        Escribir "Proyectos y tabla cargados correctamente."
    5.42    Sino
    5.43        Escribir "No se encontro el archivo ", archivo_proyectos
    5.44    FinSi
    5.45 FinSubProceso

    6.49                                    Sino
    6.50                                        Si opcion = "9" Entonces
    6.51                                            resumen_conteo_matriz()
    6.52                                        Sino
    6.53                                            Si opcion = "10" Entonces
    6.54                                                buscar_analizar_cadenas()
    6.55                                            Sino
    6.56                                                Si opcion = "11" Entonces
    6.57                                                    guardar_datos_archivos()
    6.58                                                Sino
    6.59                                                    Si opcion = "12" Entonces
    6.60                                                        cargar_datos_archivos()
    6.61                                                    Sino
    6.62                                                        Si opcion = "13" Entonces
    6.63                                                            Escribir "Cerrando programa"
    6.64                                                            ejecutando <-- Falso
    6.65                                                        Sino
    6.66                                                            Escribir "Opcion no válida. Por favor escribe un numero del 1 al 13."
    6.67                                                        FinSi
    6.68                                                    FinSi
    6.69                                                FinSi
    6.70                                            FinSi
    6.71                                        FinSi
    6.72                                    FinSi
    6.73                                FinSi
    6.74                            FinSi
    6.75                        FinSi
    6.76                    FinSi

    6.79        FinSi
    6.80        Escribir "Presiona Enter para continuar"
    6.81        Leer opcion
    6.82    FinMientras
    6.83 FinSubProceso
