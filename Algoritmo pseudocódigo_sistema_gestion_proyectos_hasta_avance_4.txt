Algoritmo sistema_gestion_proyectos

    Definir proyectos_registrados Como Cadena # [Avance 1 y 6] Crea la lista global para guardar los proyectos
    Definir matriz_proyectos Como Cadena # [Avance 7] Crea la tabla global (matriz) para guardar filas y columnas
    Definir archivo_proyectos Como Cadena # [Avance 9] Guarda el nombre del archivo de proyectos
    Definir archivo_matriz Como Cadena # [Avance 9] Guarda el nombre del archivo de la tabla

    archivo_proyectos <- "proyectos_guardados.txt" # [Avance 9] Asigna el nombre del archivo de proyectos
    archivo_matriz <- "matriz_proyectos.txt" # [Avance 9] Asigna el nombre del archivo de la tabla

    SubProceso calcular_restante (inicial, gastado) # [Avance 3] Crea la funcion para restar presupuesto
        Retornar inicial - gastado # [Avance 2 y 3] Regresa cuanto dinero queda disponible
    FinSubProceso

    SubProceso calcular_porcentaje (cumplidas, total) # [Avance 3] Crea la funcion para calcular porcentaje
        Retornar (cumplidas / total) * 100 # [Avance 2 y 3] Regresa el porcentaje de metas logradas
    FinSubProceso

    SubProceso calcular_semanas (dias) # [Avance 3] Crea la funcion para calcular semanas
        Retornar TRUNC(dias / 7) # [Avance 2 y 3] Regresa las semanas completas sin decimales
    FinSubProceso

    SubProceso calcular_dias_sobrantes (dias, semanas) # [Avance 3] Crea la funcion para dias sobrantes
        Retornar dias - (semanas * 7) # [Avance 2 y 3] Regresa los dias que quedan sueltos
    FinSubProceso

    SubProceso generar_barra_progreso (porcentaje) # [Avance 3] Crea la funcion para la barra visual
        Definir bloques Como Entero # [Avance 1] Guarda cuantos bloques llenos tendra la barra
        Si porcentaje < 0 Entonces # [Avance 4] Revisa si el porcentaje es menor que cero
            porcentaje <- 0 # [Avance 1] Ajusta el porcentaje a cero
        FinSi # [Avance 4] Cierra la revision del limite inferior
        Si porcentaje > 100 Entonces # [Avance 4] Revisa si el porcentaje pasa de 100
            porcentaje <- 100 # [Avance 1] Ajusta el porcentaje a 100
        FinSi # [Avance 4] Cierra la revision del limite superior
        bloques <- TRUNC(porcentaje / 10) # [Avance 2 y 3] Calcula cuantos bloques de 10 tocan
        Escribir "Progreso: [", bloques, " de 10 bloques completados]" # [Avance 1] Muestra la barra en pantalla
    FinSubProceso

    SubProceso evaluar_salud_presupuesto (inicial, gastado) # [Avance 3] Crea la funcion para revisar el dinero
        Definir porcentaje_gastado Como Real # [Avance 1] Guarda que porcentaje del dinero ya se gasto
        Si inicial > 0 Entonces # [Avance 4] Revisa que el presupuesto inicial sea mayor que cero
            porcentaje_gastado <- (gastado / inicial) * 100 # [Avance 2 y 3] Calcula el porcentaje gastado
        Sino # [Avance 4] Si el presupuesto inicial es cero o menos
            porcentaje_gastado <- 0 # [Avance 1] Evita dividir entre cero
        FinSi # [Avance 4] Cierra la revision del presupuesto inicial
        Si porcentaje_gastado > 90 Entonces # [Avance 4] Revisa si ya se gasto mas del 90%
            Escribir "Alerta: Presupuesto casi agotado (Mayor al 90%)" # [Avance 1] Muestra aviso de alerta
        Sino # [Avance 4] Si no pasa del 90%
            Si porcentaje_gastado >= 70 Entonces # [Avance 4] Revisa si esta entre 70% y 90%
                Escribir "Precaucion: Presupuesto en riesgo moderado (Entre 70% y 90%)" # [Avance 1] Muestra aviso de cuidado
            Sino # [Avance 4] Si es menor al 70%
                Escribir "Saludable: Presupuesto bajo control (Menor al 70%)" # [Avance 1] Muestra aviso de buen control
            FinSi # [Avance 4] Cierra la condicion intermedia
        FinSi # [Avance 4] Cierra la condicion principal
    FinSubProceso

    SubProceso crear_cuenta() # [Avance 3] Crea la funcion para registrar una cuenta
        Definir nuevo_usuario, nueva_contrasena Como Cadena # [Avance 1] Guarda usuario y contrasena nuevos
        Escribir "REGISTRO NUEVA CUENTA" # [Avance 1] Muestra el titulo de registro
        Escribir "Ingrese un nombre de usuario:" # [Avance 1] Pide el nombre de usuario
        Leer nuevo_usuario # [Avance 1] Lee el nombre de usuario
        Escribir "Ingrese una contrasena:" # [Avance 1] Pide la contrasena
        Leer nueva_contrasena # [Avance 1] Lee la contrasena
        Si nuevo_usuario <> "" Y nueva_contrasena <> "" Entonces # [Avance 4] Revisa que ambos datos esten llenos
            Escribir "Cuenta registrada con exito para: ", nuevo_usuario # [Avance 1] Confirma que la cuenta se creo
        Sino # [Avance 4] Si falta algun dato
            Escribir "Error: Debe llenar todos los campos" # [Avance 1] Avisa que faltan campos
        FinSi # [Avance 4] Cierra la revision de campos
    FinSubProceso

    SubProceso iniciar_sesion() # [Avance 3] Crea la funcion para iniciar sesion
        Definir usuario, contrasena Como Cadena # [Avance 1] Guarda usuario y contrasena ingresados

    SubProceso registrar_proyecto() # [Avance 3] Crea la funcion para registrar proyectos
        Definir nombre Como Cadena # [Avance 1] Guarda el nombre del proyecto
        Definir avance Como Real # [Avance 1] Guarda el porcentaje de avance
        Definir continuar Como Cadena # [Avance 1] Controla si se registra otro proyecto
        Definir estado_inicial Como Cadena # [Avance 7] Guarda el estado inicial del proyecto
        continuar <- "s" # [Avance 5] Empieza en "s" para entrar al ciclo
        Mientras continuar = "s" O continuar = "S" Hacer # [Avance 5] Repite el registro mientras responda "s" o "S"
            Escribir "REGISTRO DE PROYECTO" # [Avance 1] Muestra el titulo de registro de proyecto
            Escribir "Nombre del proyecto:" # [Avance 1] Pide el nombre del proyecto
            Leer nombre # [Avance 1] Lee el nombre del proyecto
            Escribir "Porcentaje de avance (0 a 100):" # [Avance 1] Pide el porcentaje de avance
            Leer avance # [Avance 1] Lee el porcentaje de avance
            Si avance >= 0 Y avance <= 100 Entonces # [Avance 4] Revisa que el avance este entre 0 y 100
                Escribir "Proyecto ", nombre, " registrado con ", avance, "% de avance" # [Avance 1] Muestra el registro hecho
                generar_barra_progreso(avance) # [Avance 3] Muestra la barra visual de avance
                Si avance = 100 Entonces # [Avance 7] Revisa si esta al 100%
                    estado_inicial <- "Completado" # [Avance 7] Marca estado como Completado
                Sino # [Avance 7] Si no esta al 100%
                    Si avance > 0 Entonces # [Avance 7] Revisa si tiene avance mayor a cero
                        estado_inicial <- "En Proceso" # [Avance 7] Marca estado como En Proceso
                    Sino # [Avance 7] Si esta en cero
                        estado_inicial <- "Iniciado" # [Avance 7] Marca estado como Iniciado
                    FinSi # [Avance 7] Cierra revision de avance mayor a cero
                FinSi # [Avance 7] Cierra revision de 100%
                AGREGAR_A_LISTA(proyectos_registrados, nombre + " - " + CONVERTIR_A_TEXTO(avance) + "%") # [Avance 6] Agrega el proyecto a la lista
                AGREGAR_A_MATRIZ(matriz_proyectos, nombre, CONVERTIR_A_TEXTO(avance) + "%", estado_inicial) # [Avance 7] Agrega la fila a la tabla
            Sino # [Avance 4] Si el numero esta fuera de 0 a 100
                Escribir "El porcentaje debe estar entre 0 y 100" # [Avance 1] Avisa que el rango no es valido
            FinSi # [Avance 4] Cierra revision del rango
            Escribir "Desea registrar otro proyecto? (s/n):" # [Avance 1] Pregunta si quiere seguir registrando
            Leer continuar # [Avance 5] Lee la respuesta para continuar o salir del ciclo
        FinMientras # [Avance 5] Cierra el ciclo de registro
    FinSubProceso

    SubProceso calcular_metricas() # [Avance 3] Crea la funcion para calcular cuentas de tiempo y dinero
        Definir nombre Como Cadena # [Avance 1] Guarda el nombre del proyecto
        Definir metas_total, metas_cumplidas Como Entero # [Avance 1] Guarda total de metas y metas logradas
        Definir presupuesto_inicial, presupuesto_gastado Como Real # [Avance 1] Guarda dinero inicial y gastado
        Definir dias, semanas, dias_s Como Entero # [Avance 1] Guarda dias totales, semanas y dias extra
        Definir presupuesto_restante, porcentaje_avance Como Real # [Avance 1] Guarda dinero restante y avance de metas
        Escribir "CALCULO DE METRICAS DE FINANZAS Y TIEMPO" # [Avance 1] Muestra el titulo de metricas
        Escribir "Nombre del proyecto:" # [Avance 1] Pide el nombre del proyecto
        Leer nombre # [Avance 1] Lee el nombre del proyecto
        Escribir "Total de metas:" # [Avance 1] Pide el total de metas
        Leer metas_total # [Avance 1] Lee el total de metas
        Escribir "Metas alcanzadas:" # [Avance 1] Pide las metas alcanzadas
        Leer metas_cumplidas # [Avance 1] Lee las metas alcanzadas
        Escribir "Presupuesto inicial:" # [Avance 1] Pide el presupuesto inicial
        Leer presupuesto_inicial # [Avance 1] Lee el presupuesto inicial
        Escribir "Presupuesto gastado:" # [Avance 1] Pide el presupuesto gastado
        Leer presupuesto_gastado # [Avance 1] Lee el presupuesto gastado
        Escribir "Duracion estimada en dias del proyecto:" # [Avance 1] Pide los dias estimados
        Leer dias # [Avance 1] Lee los dias estimados
        Si metas_total <= 0 Entonces # [Avance 4] Revisa que haya al menos 1 meta
            Escribir "Error: Sus metas deben ser al menos 1" # [Avance 1] Avisa que las metas deben ser mayores a cero
        Sino # [Avance 4] Si el total de metas es valido
            presupuesto_restante <- calcular_restante(presupuesto_inicial, presupuesto_gastado) # [Avance 2 y 3] Calcula el dinero disponible
            porcentaje_avance <- calcular_porcentaje(metas_cumplidas, metas_total) # [Avance 2 y 3] Calcula el porcentaje de metas
            semanas <- calcular_semanas(dias) # [Avance 2 y 3] Calcula las semanas completas
            dias_s <- calcular_dias_sobrantes(dias, semanas) # [Avance 2 y 3] Calcula los dias sobrantes
            Escribir "RESUMEN DE RESULTADOS" # [Avance 1] Muestra el titulo del resumen
            Escribir "Resumen de metricas: ", nombre # [Avance 1] Muestra el nombre del proyecto
            Escribir "Avance real de metas: ", porcentaje_avance, "%" # [Avance 1] Muestra el porcentaje de metas
            generar_barra_progreso(porcentaje_avance) # [Avance 3] Muestra la barra visual del avance
            evaluar_salud_presupuesto(presupuesto_inicial, presupuesto_gastado) # [Avance 3] Muestra como va el presupuesto
            Escribir "Presupuesto disponible: $", presupuesto_restante # [Avance 2] Muestra el dinero que queda
            Escribir "Tiempo estimado: ", semanas, " semanas y ", dias_s, " dias" # [Avance 2] Muestra semanas y dias extra
        FinSi # [Avance 4] Cierra la revision de metas
    FinSubProceso

    SubProceso ver_historial_proyectos() # [Avance 3] Crea la funcion para mostrar el historial

    SubProceso buscar_analizar_cadenas() # [Avance 8] Crea la funcion para buscar y revisar texto

        Escribir "BUSQUEDA Y ANALISIS DE CADENAS EN PROYECTOS" # [Avance 1 y 8] Muestra el titulo de busqueda

                Escribir "Total de caracteres: ", LONGITUD_CADENA(elemento) # [Avance 8] Muestra cuantos caracteres tiene
            FinSi # [Avance 8] Cierra la revision de coincidencia
        FinPara # [Avance 5] Cierra el ciclo de busqueda

    SubProceso guardar_datos_archivos() # [Avance 9] Crea la funcion para guardar en archivos de texto
