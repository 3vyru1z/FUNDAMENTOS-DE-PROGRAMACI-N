    SubProceso calcular_restante (inicial, gastado) // [Avance 3] Define subproceso para restar presupuesto
        Retornar inicial - gastado // [Avance 2 y 3] Retorna el resultado de la resta

    SubProceso calcular_porcentaje (cumplidas, total) // [Avance 3] Define subproceso para calcular porcentaje
        Retornar (cumplidas / total) * 100 // [Avance 2 y 3] Retorna el cálculo del porcentaje
    FinSubProceso

    SubProceso calcular_semanas (dias) // [Avance 3] Define subproceso para calcular semanas
        Retornar TRUNC(dias / 7) // [Avance 2 y 3] Retorna la división entera para obtener semanas
    FinSubProceso

    SubProceso calcular_dias_sobrantes (dias, semanas) // [Avance 3] Define subproceso para días sobrantes
        Retornar dias - (semanas * 7) // [Avance 2 y 3] Retorna la resta para obtener días restantes
    FinSubProceso

    SubProceso generar_barra_progreso (porcentaje) // [Avance 3] Define subproceso para la barra visual
        Definir bloques Como Entero // [Avance 1] Define variable entera local para los bloques
        Definir barra Como Cadena // [Avance 1] Define variable de texto local para la barra
        
        Si porcentaje < 0 Entonces // [Avance 4] Estructura de decisión: comprueba si es menor a cero
            porcentaje <- 0 // [Avance 1] Ajusta el porcentaje a cero si es negativo
        FinSi // [Avance 4] Cierra la estructura condicional
        Si porcentaje > 100 Entonces // [Avance 4] Estructura de decisión: comprueba si supera 100
            porcentaje <- 100 // [Avance 1] Ajusta el porcentaje a 100 si excede el límite
        FinSi // [Avance 4] Cierra la estructura condicional
        
        bloques <- TRUNC(porcentaje / 10) // [Avance 2 y 3] Calcula cuántos bloques pintar dividiendo entre 10
        
        Escribir "Progreso: [", bloques, " de 10 bloques completados]" // [Avance 1] Imprime simulación del progreso
    FinSubProceso

    SubProceso evaluar_salud_presupuesto (inicial, gastado) // [Avance 3] Define subproceso para evaluar dinero
        Definir porcentaje_gastado Como Real // [Avance 1] Define variable real para el gasto
        
        Si inicial > 0 Entonces // [Avance 4] Estructura de decisión: verifica que el presupuesto inicial sea mayor a cero
            porcentaje_gastado <- (gastado / inicial) * 100 // [Avance 2 y 3] Calcula el porcentaje utilizado
        Sino // [Avance 4] Estructura de decisión caso contrario si es cero
            porcentaje_gastado <- 0 // [Avance 1] Evita división por cero asignando cero
        FinSi // [Avance 4] Cierra la estructura condicional

        Si porcentaje_gastado > 90 Entonces // [Avance 4] Estructura de decisión: evalúa si supera el 90%
            Escribir "Alerta: Presupuesto casi agotado (Mayor al 90%)" // [Avance 1] Imprime mensaje de alerta crítica
        Sino  // [Avance 4] Estructura de decisión si no supera el 90%
            Si porcentaje_gastado >= 70 Entonces // [Avance 4] Estructura de decisión: evalúa si está entre 70% y 90%
                Escribir "Precaución: Presupuesto en riesgo moderado (Entre 70% y 90%)" // [Avance 1] Imprime aviso de precaución
            Sino // [Avance 4] Estructura de decisión: si es menor al 70%
                Escribir "Saludable: Presupuesto bajo control (Menor al 70%)" // [Avance 1] Imprime aviso de estado óptimo
            FinSi // [Avance 4] Cierra la estructura condicional anidada
        FinSi // [Avance 4] Cierra la estructura condicional principal
    FinSubProceso

    SubProceso crear_cuenta() // [Avance 3] Define subproceso para registro de cuentas
        Definir nuevo_usuario, nueva_contrasena Como Cadena // [Avance 1] Define variables de texto para credenciales
        
        Escribir "REGISTRO NUEVA CUENTA" // [Avance 1] Imprime título de la opción
        Escribir "Ingrese un nombre de usuario:" // [Avance 1] Solicita el nombre de usuario
        Leer nuevo_usuario // [Avance 1] Lee y almacena el usuario
        Escribir "Ingrese una contrasena:" // [Avance 1] Solicita la contraseña
        Leer nueva_contrasena // [Avance 1] Lee y almacena la contraseña
        
        Si nuevo_usuario <> "" Y nueva_contrasena <> "" Entonces // [Avance 4] Estructura de decisión valida que no estén vacíos
            Escribir "Cuenta registrada con exito para: ", nuevo_usuario // [Avance 1] Imprime confirmación de éxito
        Sino // [Avance 4] Estructura de decisión: se ejecuta si falta algún campo
            Escribir "Error: Debe llenar todos los campos" // [Avance 1] Imprime mensaje de error
        FinSi // [Avance 4] Cierra la estructura condicional
    FinSubProceso

    SubProceso iniciar_sesion() // [Avance 3] Define subproceso para inicio de sesión
        Definir usuario, contrasena Como Cadena // [Avance 1] Define variables de texto locales
        
        Escribir "INICIO DE SESION" // [Avance 1] Imprime título de inicio de sesión
        Escribir "Ingrese su usuario:" // [Avance 1] Solicita el usuario
        Leer usuario // [Avance 1] Lee el usuario ingresado
        Escribir "Ingrese contrasena:" // [Avance 1] Solicita la contraseña
        Leer contrasena // [Avance 1] Lee la contraseña ingresada
        
        Si usuario <> "" Y contrasena <> "" Entonces // [Avance 4] Estructura de decisión comprueba campos llenos
            Escribir "Bienvenido/a, ", usuario // [Avance 1] Imprime bienvenida
        Sino // [Avance 4] Estructura de decisión se ejecuta si hay campos vacíos
            Escribir "Datos incorrectos" // [Avance 1] Imprime error de datos
        FinSi // [Avance 4] Cierra la estructura condicional
    FinSubProceso

    SubProceso registrar_proyecto() // [Avance 3] Define subproceso para registrar proyectos
        Definir nombre Como Cadena // [Avance 1] Define variable de texto para el nombre
        Definir avance Como Real // [Avance 1] Define variable real para el porcentaje
        Definir continuar Como Cadena // [Avance 1] Define variable para controlar el ciclo de repetición
        
        continuar <- "s" // [Avance 5] Inicializa la variable de control en "s" para entrar al ciclo
        
        Mientras continuar = "s" O continuar = "S" Hacer // [Avance 5] Estructura de repetición: ciclo para registrar múltiples proyectos de forma continua
            Escribir "REGISTRO DE PROYECTO" // [Avance 1] Imprime título del módulo
            Escribir "Nombre del proyecto:" // [Avance 1] Solicita el nombre del proyecto
            Leer nombre // [Avance 1] Lee el nombre ingresado
            Escribir "Porcentaje de avance (0 a 100):" // [Avance 1] Solicita el avance numérico
            Leer avance // [Avance 1] Lee el porcentaje de avance
            
            Si avance >= 0 Y avance <= 100 Entonces // [Avance 4] Estructura de decisión: valida el rango correcto
                Escribir "Proyecto ", nombre, " registrado con ", avance, "% de avance" // [Avance 1] Confirma el registro
                generar_barra_progreso(avance) // [Avance 3] Llama al subproceso de la barra gráfica
            Sino // [Avance 4] Estructura de decisión: se ejecuta si está fuera de rango
                Escribir "El porcentaje debe estar entre 0 y 100" // [Avance 1] Muestra advertencia de rango válido
            FinSi // [Avance 4] Cierra la estructura condicional
            
            Escribir "¿Desea registrar otro proyecto? (s/n):" // [Avance 1] Pregunta si desea repetir la captura
            Leer continuar // [Avance 5] Lee la respuesta para decidir si el ciclo se repite o termina
        FinMientras // [Avance 5] Cierra la estructura de repetición del registro continuo
    FinSubProceso

    SubProceso calcular_metricas() // [Avance 3] Define subproceso para calcular métricas
        Definir nombre Como Cadena // [Avance 1] Define variable de texto para el nombre
        Definir metas_total, metas_cumplidas Como Entero // [Avance 1] Define enteros para las metas
        Definir presupuesto_inicial, presupuesto_gastado Como Real // [Avance 1] Define reales para el presupuesto
        Definir dias, semanas, dias_s Como Entero // [Avance 1] Define enteros para el tiempo
        Definir presupuesto_restante, porcentaje_avance Como Real // [Avance 1] Define reales para resultados
        
        Escribir "CALCULO DE METRICAS DE FINANZAS Y TIEMPO" // [Avance 1] Imprime título de métricas
        Escribir "Nombre del proyecto:" // [Avance 1] Solicita el nombre
        Leer nombre // [Avance 1] Lee el nombre
        Escribir "Total de metas:" // [Avance 1] Solicita el total de metas
        Leer metas_total // [Avance 1] Lee el total de metas
        Escribir "Metas alcanzadas:" // [Avance 1] Solicita metas cumplidas
        Leer metas_cumplidas // [Avance 1] Lee metas cumplidas
        Escribir "Presupuesto inicial:" // [Avance 1] Solicita presupuesto inicial
        Leer presupuesto_inicial // [Avance 1] Lee presupuesto inicial
        Escribir "Presupuesto gastado:" // [Avance 1] Solicita presupuesto gastado
        Leer presupuesto_gastado // [Avance 1] Lee presupuesto gastado
        Escribir "Duracion estimada en dias del proyecto:" // [Avance 1] Solicita días estimados
        Leer dias // [Avance 1] Lee los días
        
        Si metas_total <= 0 Entonces // [Avance 4] Estructura de decisión comprueba que las metas sean válidas
            Escribir "Error: Sus metas deben ser al menos 1" // [Avance 1] Imprime error por metas nulas
        Sino // [Avance 4] Estructura de decisión: se ejecuta si las metas son correctas
            presupuesto_restante <- calcular_restante(presupuesto_inicial, presupuesto_gastado) // [Avance 2 y 3] Llama subproceso de resta
            porcentaje_avance <- calcular_porcentaje(metas_cumplidas, metas_total) // [Avance 2 y 3] Llama subproceso de porcentaje
            semanas <- calcular_semanas(dias) // [Avance 2 y 3] Llama subproceso de semanas
            dias_s <- calcular_dias_sobrantes(dias, semanas) // [Avance 2 y 3] Llama subproceso de días sobrantes
            
            Escribir "RESUMEN DE RESULTADOS" // [Avance 1] Imprime encabezado del reporte
            Escribir "Resumen de metricas: ", nombre // [Avance 1] Muestra el nombre evaluado
            Escribir "Avance real de metas: ", porcentaje_avance, "%" // [Avance 1] Imprime el porcentaje
            generar_barra_progreso(porcentaje_avance) // [Avance 3] Muestra la barra visual
            evaluar_salud_presupuesto(presupuesto_inicial, presupuesto_gastado) // [Avance 3] Muestra el estado financiero
            Escribir "Presupuesto disponible: $", presupuesto_restante // [Avance 2] Imprime saldo restante
            Escribir "Tiempo estimado: ", semanas, " semanas y ", dias_s, " dias" // [Avance 2] Imprime tiempo desglosado
        FinSi // [Avance 4] Cierra la estructura condicional
    FinSubProceso

                Sino // [Avance 4] Continuación de decisión múltiple
                    Si opcion = "3" Entonces // [Avance 4] Estructura de decisión evalúa si se eligió la opción 3

                    Sino // [Avance 4] Continuación de decisión múltiple
                        Si opcion = "4" Entonces // [Avance 4] Estructura de decisión evalúa si se eligió la opción 4
                            calcular_metricas() // [Avance 3] Llama al subproceso de cálculo de métricas
                        Sino // [Avance 4] Continuación de decisión múltiple
                            Si opcion = "5" Entonces // [Avance 4] Estructura de decisión evalúa si se eligió la opción 5
                                ver_historial_proyectos() // [Avance 3] Llama al subproceso de historial
                            Sino // [Avance 4] Continuación de decisión múltiple
                                Si opcion = "6" Entonces // [Avance 4] Estructura de decisión evalúa si se eligió la opción 6
                                    Escribir "Cerrando programa" // [Avance 1] Imprime mensaje de salida
                                    ejecutando <- Falso // [Avance 1] Cambia a Falso para romper el ciclo Mientras del menú
                                Sino // [Avance 4] Estructura de decisión caso por defecto para opción inválida
                                    Escribir "Opcion no valida. Por favor escribe un numero del 1 al 6." // [Avance 1] Muestra error de opción incorrecta
                                FinSi // [Avance 4] Cierra el Si de la opción 6
                            FinSi // [Avance 4] Cierra el Si de la opción 5
                        FinSi // [Avance 4] Cierra el Si de la opción 4
                    FinSi // [Avance 4] Cierra el Si de la opción 3
                FinSi // [Avance 4] Cierra el Si de la opción 2
            FinSi // [Avance 4] Cierra el Si de la opción 1
        FinMientras // [Avance 5] Cierra la estructura de repetición principal del menú
    FinSubProceso

    // Inicio del flujo del algoritmo
    principal() // [Avance 3] Llama al subproceso principal para poner en marcha el sistema
