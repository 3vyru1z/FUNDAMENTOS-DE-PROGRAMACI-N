    FinSubProceso

    SubProceso calcular_semanas (dias) // [Avance 3] Define subproceso para calcular semanas
        Retornar TRUNC(dias / 7) // [Avance 2 y 3] Retorna la división entera para obtener semanas
    FinSubProceso

    SubProceso calcular_dias_sobrantes (dias, semanas) // [Avance 3] Define subproceso para días sobrantes

    FinSubProceso

    SubProceso generar_barra_progreso (porcentaje) // [Avance 3] Define subproceso para la barra visual
        Definir bloques como entero // [Avance 1] Define variable entera local para los bloques
        Definir barra como cadena // [Avance 1] Define variable de texto local para la barra
        
        Si porcentaje < 0 Entonces // [Avance 4] Estructura de decisión: comprueba si es menor a cero
            porcentaje <- 0 // [Avance 1] Ajusta el porcentaje a cero si es negativo
        FinSi // [Avance 4] Cierra la estructura condicional
        Si porcentaje > 100 Entonces // [Avance 4] Estructura de decisión: comprueba si supera 100
            porcentaje <- 100 // [Avance 1] Ajusta el porcentaje a 100 si excede el límite
        FinSi // [Avance 4] Cierra la estructura condicional
        
        bloques <- TRUNC(porcentaje / 10) // [Avance 2 y 3] Calcula cuántos bloques pintar dividiendo entre 10
        
        Escribir "Progreso: [", bloques, " de 10 bloques completados]" // [Avance 1] Imprime simulación del progreso
    FinSubProceso

    SubProceso evaluar_salud_presupuesto (inicial, gastado) // [Avance 3] Define subproceso para evaluar dinero
        Definir porcentaje_gastado Como Real // [Avance 1] Define variable real para el gasto
        
        Si inicial > 0 Entonces // [Avance 4] Estructura de decisión: verifica que el presupuesto inicial sea mayor a cero
            porcentaje_gastado <- (gastado / inicial) * 100 // [Avance 2 y 3] Calcula el porcentaje utilizado
        Sino // [Avance 4] Estructura de decisión caso contrario si es cero
            porcentaje_gastado <- 0 // [Avance 1] Evita división por cero asignando cero
        FinSi // [Avance 4] Cierra la estructura condicional

        Si porcentaje_gastado > 90 Entonces // [Avance 4] Estructura de decisión: evalúa si supera el 90%
            Escribir "Alerta: Presupuesto casi agotado (Mayor al 90%)" // [Avance 1] Imprime mensaje de alerta crítica
        Sino  // [Avance 4] Estructura de decisión si no supera el 90%
            Si porcentaje_gastado >= 70 Entonces // [Avance 4] Estructura de decisión: evalúa si está entre 70% y 90%
                Escribir "Precaución: Presupuesto en riesgo moderado (Entre 70% y 90%)" // [Avance 1] Imprime aviso de precaución
            Sino // [Avance 4] Estructura de decisión: si es menor al 70%
                Escribir "Saludable: Presupuesto bajo control (Menor al 70%)" // [Avance 1] Imprime aviso de estado óptimo
            FinSi // [Avance 4] Cierra la estructura condicional anidada
        FinSi // [Avance 4] Cierra la estructura condicional principal
    FinSubProceso

    SubProceso crear_cuenta() // [Avance 3] Define subproceso para registro de cuentas

            Escribir "Cuenta registrada con exito para: ", nuevo_usuario // [Avance 1] Imprime confirmación de éxito
        Sino // [Avance 4] Estructura de decisión: se ejecuta si falta algún campo

    SubProceso iniciar_sesion() // [Avance 3] Define subproceso para inicio de sesión

    SubProceso registrar_proyecto() // [Avance 3] Define subproceso para registrar proyectos
        Definir nombre como cadena // [Avance 1] Define variable de texto para el nombre
        Definir avance como real // [Avance 1] Define variable real para el porcentaje
        Definir continuar como cadena // [Avance 1] Define variable para controlar el ciclo de repetición
        
        continuar  "s" // [Avance 5] Inicializa la variable de control en "s" para entrar al ciclo
        
        Mientras continuar = "s" O continuar = "S" Hacer // [Avance 5] Estructura de repetición: ciclo para registrar múltiples proyectos de forma continua
            Escribir "REGISTRO DE PROYECTO" // [Avance 1] Imprime título del módulo
            Escribir "Nombre del proyecto:" // [Avance 1] Solicita el nombre del proyecto
            Leer nombre // [Avance 1] Lee el nombre ingresado
            Escribir "Porcentaje de avance (0 a 100):" // [Avance 1] Solicita el avance numérico
            Leer avance // [Avance 1] Lee el porcentaje de avance
            
            Si avance >= 0 Y avance <= 100 Entonces // [Avance 4] Estructura de decisión: valida el rango correcto
                Escribir "Proyecto ", nombre, " registrado con ", avance, "% de avance" // [Avance 1] Confirma el registro
                generar_barra_progreso(avance) // [Avance 3] Llama al subproceso de la barra gráfica
                // [Avance 6] Operación de arreglos simples
                AGREGAR_A_LISTA(proyectos_registrados, nombre + " - " + CONVERTIR_A_TEXTO(avance) + "%") 
                // [Avance 7] Operación de listas anidadas (Matriz): Agrega fila con atributos tabulares
                AGREGAR_A_MATRIZ(matriz_proyectos, nombre, CONVERTIR_A_TEXTO(avance) + "%", "Activo")
            Sino // [Avance 4] Estructura de decisión: se ejecuta si está fuera de rango
                Escribir "El porcentaje debe estar entre 0 y 100" // [Avance 1] Muestra advertencia de rango válido
            FinSi // [Avance 4] Cierra la estructura condicional
            
            Escribir "¿Desea registrar otro proyecto? (s/n):" // [Avance 1] Pregunta si desea repetir la captura
            Leer continuar // [Avance 5] Lee la respuesta para decidir si el ciclo se repite o termina
        FinMientras // [Avance 5] Cierra la estructura de repetición del registro continuo
    FinSubProceso

    SubProceso calcular_metricas() // [Avance 3] Define subproceso para calcular métricas

        Leer metas_cumplidas // [Avance 1] Lee metas cumplidas

        Leer presupuesto_inicial // [Avance 1] Lee presupuesto inicial
        Escribir "Presupuesto gastado:" // [Avance 1] Solicita presupuesto gastado

        Sino // [Avance 4] Estructura de decisión: se ejecuta si las metas son correctas
            presupuesto_restante <- calcular_restante(presupuesto_inicial, presupuesto_gastado) // [Avance 2 y 3] Llama subproceso de resta

            Escribir "Avance real de metas: ", porcentaje_avance, "%" // [Avance 1] Imprime el porcentaje
